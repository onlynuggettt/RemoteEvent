Exchanging..."); const response = await fetch( `${BACKEND_URL}/api/exchange?ticket=${encodeURIComponent(ticket)}`, { method: "GET", cache: "no-store" } ); const data = await response.json(); console.log("Ticket exchange:", data); if (data.loggedIn && data.user) { showAccount(data.user); // Remove ticket from URL window.history.replaceState( {}, document.title, window.location.pathname ); return; } console.error("Ticket exchange failed:", data); } // ========================================== // NORMAL SESSION CHECK // ========================================== const response = await fetch( `${BACKEND_URL}/api/me`, { method: "GET", credentials: "include", cache: "no-store" } ); const data = await response.json(); console.log("RemoteEvent login:", data); if (data.loggedIn && data.user) { showAccount(data.user); } else { showLoggedOut(); } } catch (error) { console.error("Login check failed:", error); showLoggedOut(); } } // ========================================== // SHOW ACCOUNT // ========================================== function showAccount(user) { const loginButton = document.getElementById("loginButton"); const heroLoginButton = document.getElementById("heroLoginButton"); const accountSection = document.getElementById("accountSection"); const avatar = document.getElementById("accountAvatar"); const username = document.getElementById("accountUsername"); const displayName = document.getElementById("accountDisplayName"); const userId = document.getElementById("accountUserId"); if (loginButton) { loginButton.style.display = "none"; } if (heroLoginButton) { heroLoginButton.style.display = "none"; } if (accountSection) { accountSection.style.display = "block"; } if (avatar) { avatar.src = user.picture || ""; } if (username) { username.textContent = "@" + user.username; } if (displayName) { displayName.textContent = user.displayName; const heroLoginButton = document.getElementById("heroLoginButton"); const accountSection = document.getElementById("accountSection"); if (accountSection) { accountSection.style.display = "none"; } if (loginButton) { loginButton.style.display = ""; } if (heroLoginButton) { heroLoginButton.style.display = ""; } } // ========================================== // LOGOUT // ========================================== async function logout() { try { await fetch(`${BACKEND_URL}/logout`, { method: "GET", credentials: "include" });
